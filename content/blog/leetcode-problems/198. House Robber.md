```python
class Solution:
    def rob(self, nums: List[int]) -> int:
        n = len(nums)

        @lru_cache(None)
        def robHelper(idx: int, in_cooldown: bool) -> int:
            if idx >= n:
                return 0
            if in_cooldown:
                return robHelper(idx + 1, False)
            return max(
                (nums[idx] + robHelper(idx + 1, True)), robHelper(idx + 1, False)
            )

        return robHelper(0, False)
```
