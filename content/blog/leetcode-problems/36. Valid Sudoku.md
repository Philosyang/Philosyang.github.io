```python
# primitive idea
# use sets, check before .add
# go over each row
# go over each col
# go over each 3x3


class Solution:
    def isValidSudoku(self, board: list[list[str]]) -> bool:
        # assert board
        m, n = len(board), len(board[0])
        # per row
        for i in range(m):
            values = set()
            for j in range(n):
                if board[i][j] == ".":
                    continue
                elif board[i][j] in values:
                    return False
                else:
                    values.add(board[i][j])
        # per col
        for j in range(n):
            values = set()
            for i in range(m):
                if board[i][j] == ".":
                    continue
                elif board[i][j] in values:
                    return False
                else:
                    values.add(board[i][j])

        # per 3x3
        def scan33(a: int, b: int) -> bool:
            values = set()
            for i in range(a, a + 3):
                for j in range(b, b + 3):
                    if board[i][j] == ".":
                        continue
                    elif board[i][j] in values:
                        return False
                    else:
                        values.add(board[i][j])
            return True

        for i in range(m):
            for j in range(n):
                if (i % 3 == 0) and (j % 3 == 0):
                    if not scan33(i, j):
                        return False

        return True
```
