```python
# idea:
# { prereq: [course0, course1, ...] } dictionary, dfs + memo (everything after this prereq works)

class Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        good_courses = set()
        prereqs = defaultdict(list)
        for a, b in prerequisites:
            prereqs[a].append(b)

        def search(course: int, parents: set) -> bool:
            if course in good_courses or not prereqs[course]:
                return True

            for child in prereqs[course]:
                if child in parents:
                    return False
                parents.add(child)
                result = search(child, parents)
                parents.remove(child)
                if not result:
                    return False
                good_courses.add(child)

            return True

        for i in range(numCourses):
            if not search(i, set()):
                return False

        return True
```
