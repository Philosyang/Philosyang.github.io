```python
class Solution:
    def orangesRotting(self, grid: List[List[int]]) -> int:
        # impossible == fresh left when bfs finishes
        # level-based bfs

        fresh_count = 0
        rotten = []
        directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]
        minutes = 0

        m, n = len(grid), len(grid[0])
        for i in range(m):
            for j in range(n):
                current_cell = grid[i][j]
                if current_cell == 1:
                    fresh_count += 1
                elif current_cell == 2:
                    rotten.append((i, j))

        while rotten and fresh_count > 0:
            minutes += 1
            adjacent = []
            for x, y in rotten:
                for dx, dy in directions:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < m and 0 <= ny < n and grid[nx][ny] == 1:
                        grid[nx][ny] = 2
                        fresh_count -= 1
                        adjacent.append((nx, ny))
            rotten = adjacent

        return -1 if fresh_count else minutes
```
