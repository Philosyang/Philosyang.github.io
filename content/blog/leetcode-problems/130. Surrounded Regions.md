```python
# idea:
# don't: linear scan from (1,1) to (m-2,n-2), if 'O', update while bfs
# DO: iterate through the outer rim, update & bfs all 'O' regions into '*'
# then flip everything 'O' into 'X'
# then flip everything '*' into 'O'


class Solution:
    def solve(self, board: List[List[str]]) -> None:
        """
        Do not return anything, modify board in-place instead.
        """
        m, n = len(board), len(board[0])
        dirs = [(0, 1), (0, -1), (1, 0), (-1, 0)]

        def search(x: int, y: int) -> None:
            if board[x][y] == "O":
                board[x][y] = "*"
                for dx, dy in dirs:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < m and 0 <= ny < n:
                        search(nx, ny)

        # traverse outer rim
        for i in range(m):
            for j in range(n):
                if i in [0, m - 1] or j in [0, n - 1]:
                    search(i, j)

        # flip everything still 'O' into 'X'
        for i in range(m):
            for j in range(n):
                if board[i][j] == "O":
                    board[i][j] = "X"

        # flip everything '*' into 'O'
        for i in range(m):
            for j in range(n):
                if board[i][j] == "*":
                    board[i][j] = "O"
```
