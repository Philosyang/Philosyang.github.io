```python
class Solution:
    def solve(self, board: list[list[str]]) -> None:
        """
        Do not return anything, modify board in-place instead.
        """
        # idea: two-pass
        # first mark all 'O's on the edge to 'T'
        # then traversal and keep history and mark "uncaptureable" cells to skip
        # for each traverse: keep a traversed set and a bool flag
        # if has 'T', mark all uncaptureable/skipped
        # if no 'T', overwrite all into 'X'

        m, n = len(board), len(board[0])
        uncaptureables = set()
        traversed = set()
        has_T = False
        directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]

        for i in range(m):
            for j in range(n):
                if (i in [0, m - 1] or j in [0, n - 1]) and board[i][j] == "O":
                    board[i][j] = "T"

        def traversal(a: int, b: int) -> None:
            nonlocal has_T
            if (a, b) in uncaptureables or (a, b) in traversed or board[a][b] == "X":
                return
            traversed.add((a, b))
            if board[a][b] == "T":
                has_T = True
            for dx, dy in directions:
                nx, ny = a + dx, b + dy
                if 0 <= nx < m and 0 <= ny < n:
                    traversal(nx, ny)

        for i in range(m):
            for j in range(n):
                traversed = set()
                has_T = False
                traversal(i, j)
                for r, s in traversed:
                    if has_T:
                        uncaptureables.add((r, s))
                    else:
                        board[r][s] = "X"
        for i in range(m):
            for j in range(n):
                if board[i][j] == "T":
                    board[i][j] = "O"

        return
```
