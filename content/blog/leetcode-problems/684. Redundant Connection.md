```python
# keep a ultimate-parents dict, update answer as we scan through edges
# answer is updated whenever "this node already has this parent"
# before dealing with this edge
# e.g. for [[1,2],[2,3],[3,4],[1,4],[1,5]]
# [3,4] puts {4: 1} in our parents dict
# when we see [1,4], since 4 already has 1 as the parent, answer = [1,4]
# note that you need to trace both 4's parent and 1's parent


class Solution:
    def findRedundantConnection(self, edges: List[List[int]]) -> List[int]:
        n = len(edges)
        parents = {x: x for x in range(1, n + 1)}  # you are your parent
        children = {x: {x} for x in range(1, n + 1)}
        answer = None

        for edge in edges:
            a, b = edge
            pa, pb = parents[a], parents[b]
            if pa == pb:  # share same parent
                answer = edge
            new_parent = min(pa, pb)
            parents[a] = new_parent
            parents[b] = new_parent
            update_child = max(pa, pb)
            if pa != pb:
                for child in children[update_child]:
                    parents[child] = new_parent
                    children[new_parent].add(child)
                children[update_child] = None
            # print(parents)
            # print(children)

        return answer
```
